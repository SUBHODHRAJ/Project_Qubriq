using System.Security.Claims;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using Qubriq.API.Data;
using Qubriq.API.DTOs;
using Qubriq.API.Models;

namespace Qubriq.API.Controllers;

[ApiController]
[Authorize]
[Route("api/bookings")]
public class BookingsController : ControllerBase
{
    private readonly QubriqDbContext _db;

    public BookingsController(QubriqDbContext db)
    {
        _db = db;
    }

    private long? GetAuthenticatedUserId()
    {
        var subject =
            User.FindFirst("sub")?.Value ??
            User.FindFirst(ClaimTypes.NameIdentifier)?.Value;

        if (long.TryParse(subject, out var userId))
        {
            return userId;
        }

        return null;
    }

    private async Task<bool> HasPermissionAsync(
        long userId,
        string permissionCode)
    {
        return await _db.UserRoles
            .Where(userRole => userRole.UserId == userId)
            .Join(
                _db.RolePermissions,
                userRole => userRole.RoleId,
                rolePermission => rolePermission.RoleId,
                (userRole, rolePermission) => rolePermission)
            .AnyAsync(rolePermission =>
                rolePermission.IsAllowed &&
                rolePermission.Permission.Code == permissionCode);
    }

    // GET: api/bookings
    // Internal users see all permitted bookings.
    // Booking customers see only their own bookings.
    [HttpGet]
    [Authorize(Policy = "BookingsRead")]
    public async Task<IActionResult> GetBookings(
        [FromQuery] string? status,
        [FromQuery] string? search)
    {
        var userId = GetAuthenticatedUserId();

        if (!userId.HasValue)
        {
            return Unauthorized();
        }

        var canReadAll = await HasPermissionAsync(
            userId.Value,
            "bookings.read_all");

        var query = _db.Shipments
            .AsNoTracking()
            .Include(x => x.Customer)
            .AsQueryable();

        if (!canReadAll)
        {
            query = query.Where(x =>
                x.Customer.UserId == userId.Value);
        }

        if (!string.IsNullOrWhiteSpace(status) &&
            !status.Equals("All", StringComparison.OrdinalIgnoreCase))
        {
            query = query.Where(x => x.Status == status);
        }

        if (!string.IsNullOrWhiteSpace(search))
        {
            var term = search.Trim();

            query = query.Where(x =>
                x.TrackingNumber.Contains(term) ||
                x.Customer.CompanyName.Contains(term));
        }

        var bookings = await query
            .OrderByDescending(x => x.CreatedAtUtc)
            .Select(x => new
            {
                x.Id,
                bookingId = x.TrackingNumber,
                customer = x.Customer.CompanyName,
                pickupLocation = x.Origin,
                deliveryLocation = x.Destination,
                cargoType = x.CargoCategory,
                cargoName = x.CargoName,
                weight = x.TotalWeightKg,
                x.PackageCount,
                x.Priority,
                bookingDate = x.CreatedAtUtc,
                pickupDate = x.PickupDateUtc,
                expectedDeliveryDate = x.ExpectedDeliveryUtc,
                x.Status
            })
            .ToListAsync();

        return Ok(bookings);
    }

    // GET: api/bookings/customers
    // Only users with bookings.read_all can list customer records.
    [HttpGet("customers")]
    [Authorize(Policy = "BookingsReadAll")]
    public async Task<IActionResult> GetCustomers()
    {
        var customers = await _db.Customers
            .AsNoTracking()
            .Where(x => x.IsActive)
            .OrderBy(x => x.CompanyName)
            .Select(x => new
            {
                x.Id,
                x.CompanyName,
                x.ContactName,
                x.Email,
                x.Phone,
                x.Address
            })
            .ToListAsync();

        return Ok(customers);
    }

    // POST: api/bookings
    [HttpPost]
    [Authorize(Policy = "BookingsCreate")]
    public async Task<IActionResult> CreateBooking(
        [FromBody] CreateBookingRequest request)
    {
        var userId = GetAuthenticatedUserId();

        if (!userId.HasValue)
        {
            return Unauthorized();
        }

        var canReadAll = await HasPermissionAsync(
            userId.Value,
            "bookings.read_all");

        Customer? customer = null;

        if (canReadAll)
        {
            if (!request.CustomerId.HasValue ||
                request.CustomerId.Value <= 0)
            {
                return BadRequest(new
                {
                    message = "A valid customer must be selected."
                });
            }

            customer = await _db.Customers
                .FirstOrDefaultAsync(x =>
                    x.Id == request.CustomerId.Value &&
                    x.IsActive);
        }
        else
        {
            // Never trust a customer ID submitted by an external customer.
            // Resolve the customer's own profile from the authenticated user.
            customer = await _db.Customers
                .FirstOrDefaultAsync(x =>
                    x.UserId == userId.Value &&
                    x.IsActive);
        }

        if (customer == null)
        {
            return BadRequest(new
            {
                message = canReadAll
                    ? "Select an active customer before creating a booking."
                    : "Your account does not have an active customer profile. Contact Qubriq support."
            });
        }

        if (request.PickupDateUtc == default)
        {
            return BadRequest(new
            {
                message = "A valid pickup date is required."
            });
        }

        if (request.ExpectedDeliveryUtc.HasValue &&
            request.ExpectedDeliveryUtc.Value < request.PickupDateUtc)
        {
            return BadRequest(new
            {
                message = "Expected delivery cannot be earlier than pickup."
            });
        }

        var trackingNumber =
            $"QBR-{DateTime.UtcNow:yyyyMMdd}-" +
            Guid.NewGuid().ToString("N")[..8].ToUpperInvariant();

        var shipment = new Shipment
        {
            TrackingNumber = trackingNumber,
            CustomerId = customer.Id,
            Origin = request.Origin.Trim(),
            Destination = request.Destination.Trim(),
            Status = request.Status,
            CreatedAtUtc = DateTime.UtcNow,
            ExpectedDeliveryUtc = request.ExpectedDeliveryUtc,

            CargoName = request.CargoName.Trim(),
            CargoCategory = request.CargoCategory?.Trim(),
            PackageCount = request.PackageCount,
            TotalWeightKg = request.TotalWeightKg,
            LengthCm = request.LengthCm,
            WidthCm = request.WidthCm,
            HeightCm = request.HeightCm,
            IsFragile = request.IsFragile,
            SpecialHandlingRequirements =
                request.SpecialHandlingRequirements?.Trim(),

            Priority = request.Priority,
            PickupDateUtc = request.PickupDateUtc,
            DeliveryTimeWindow = request.DeliveryTimeWindow?.Trim(),
            AdditionalInstructions =
                request.AdditionalInstructions?.Trim()
        };

        _db.Shipments.Add(shipment);

        await _db.SaveChangesAsync();

        return CreatedAtAction(
            nameof(GetBookings),
            new { },
            new
            {
                shipment.Id,
                shipment.TrackingNumber,
                shipment.Status,
                message = "Booking created successfully."
            });
    }
}

                    User.FindFirst(ClaimTypes.NameIdentifier)?.Value;

        return long.TryParse(value, out var id) ? id : null;
    }

    private async Task<bool> HasPermission(string code)
    {
        var id = AdminId();
        if (!id.HasValue) return false;

        return await _db.UserRoles
            .Where(x => x.UserId == id.Value)
            .Join(_db.RolePermissions,
                x => x.RoleId,
                x => x.RoleId,
                (_, p) => p)
            .AnyAsync(x => x.IsAllowed && x.Permission.Code == code);
    }

    public sealed class DriverAssignmentRequest
    {
        public long ShipmentId { get; set; }
        public long DriverId { get; set; }
        public long VehicleId { get; set; }
    }

    [HttpPost("driver")]
    public async Task<IActionResult> AssignDriver(
        [FromBody] DriverAssignmentRequest request)
    {
        if (!await HasPermission("delivery.assignments.manage"))
            return Forbid();

        await using var transaction =
            await _db.Database.BeginTransactionAsync(IsolationLevel.Serializable);

        var shipment = await _db.Shipments
            .FirstOrDefaultAsync(x => x.Id == request.ShipmentId);

        if (shipment == null)
            return NotFound(new { message = "Shipment not found." });

        var driver = await _db.Drivers
            .FirstOrDefaultAsync(x => x.Id == request.DriverId);

        var vehicle = await _db.Vehicles
            .FirstOrDefaultAsync(x => x.Id == request.VehicleId);

        if (driver == null || vehicle == null)
            return BadRequest(new { message = "Valid driver and vehicle are required." });

        if (driver.Status != "Available" || vehicle.Status != "Available")
            return Conflict(new { message = "Driver and vehicle must be Available." });

        var shipmentAlreadyAssigned = await _db.ShipmentAssignments.AnyAsync(x =>
            x.ShipmentId == request.ShipmentId &&
            x.Status == "Active" &&
            x.ReleasedAtUtc == null);

        if (shipmentAlreadyAssigned)
            return Conflict(new { message = "Shipment already has an active assignment." });

        var driverAlreadyAssigned = await _db.ShipmentAssignments.AnyAsync(x =>
            x.DriverId == request.DriverId &&
            x.Status == "Active" &&
            x.ReleasedAtUtc == null);

        if (driverAlreadyAssigned)
            return Conflict(new { message = "Driver already has an active assignment." });

        var vehicleAlreadyAssigned = await _db.ShipmentAssignments.AnyAsync(x =>
            x.VehicleId == request.VehicleId &&
            x.Status == "Active" &&
            x.ReleasedAtUtc == null);

        if (vehicleAlreadyAssigned)
            return Conflict(new { message = "Vehicle already has an active assignment." });

        var assignment = new ShipmentAssignment
        {
            ShipmentId = shipment.Id,
            DriverId = driver.Id,
            VehicleId = vehicle.Id,
            AssignedAtUtc = DateTime.UtcNow,
            Status = "Active"
        };

        _db.ShipmentAssignments.Add(assignment);

        shipment.Status = "Assigned";

        _db.AuditLogs.Add(new AuditLog
        {
            UserId = AdminId(),
            Action = "SHIPMENT_DRIVER_ASSIGNED",
            EntityType = "Shipment",
            EntityId = shipment.Id.ToString(),
            Details = $"AssignmentId={assignment.Id}; DriverId={driver.Id}; VehicleId={vehicle.Id}",
            CreatedAtUtc = DateTime.UtcNow
        });

        await _db.SaveChangesAsync();
        await transaction.CommitAsync();

        return Ok(new
        {
            assignment.Id,
            assignment.ShipmentId,
            assignment.DriverId,
            assignment.VehicleId,
            shipment.Status
        });
    }

    public sealed class FinalRecipientRequest
    {
        public long ShipmentId { get; set; }
        public long RecipientUserId { get; set; }
    }

    [HttpPost("recipient")]
    public async Task<IActionResult> AssignFinalRecipient(
        [FromBody] FinalRecipientRequest request)
    {
        if (!await HasPermission("delivery.assignments.manage"))
            return Forbid();

        await using var transaction =
            await _db.Database.BeginTransactionAsync(IsolationLevel.Serializable);

        var user = await _db.Users
            .FirstOrDefaultAsync(x =>
                x.Id == request.RecipientUserId &&
                x.IsActive &&
                x.RecipientType == "FinalRecipient");

        if (user == null)
            return BadRequest(new { message = "Active FinalRecipient account not found." });

        var shipmentExists = await _db.Shipments
            .AnyAsync(x => x.Id == request.ShipmentId);

        if (!shipmentExists)
            return NotFound(new { message = "Shipment not found." });

        if (await _db.ShipmentRecipients.AnyAsync(x =>
            x.ShipmentId == request.ShipmentId &&
            x.UserId == request.RecipientUserId))
            return Conflict(new { message = "Recipient is already linked to this shipment." });

        _db.ShipmentRecipients.Add(new ShipmentRecipient
        {
            ShipmentId = request.ShipmentId,
            UserId = request.RecipientUserId,
            CreatedAtUtc = DateTime.UtcNow
        });

        _db.AuditLogs.Add(new AuditLog
        {
            UserId = AdminId(),
            Action = "FINAL_RECIPIENT_ASSIGNED",
            EntityType = "Shipment",
            EntityId = request.ShipmentId.ToString(),
            Details = $"RecipientUserId={request.RecipientUserId}",
            CreatedAtUtc = DateTime.UtcNow
        });

        await _db.SaveChangesAsync();
        await transaction.CommitAsync();

        return Ok(new { message = "Final recipient linked to shipment." });
    }
}

using System.Security.Claims;
using Microsoft.AspNetCore.Authorization;
using Microsoft.EntityFrameworkCore;
using Qubriq.API.Data;

namespace Qubriq.API.Authorization;

public class PermissionAuthorizationHandler
    : AuthorizationHandler<PermissionRequirement>
{
    private readonly QubriqDbContext _db;

    public PermissionAuthorizationHandler(QubriqDbContext db)
    {
        _db = db;
    }

    protected override async Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        PermissionRequirement requirement)
    {
        var subject =
            context.User.FindFirst("sub")?.Value ??
            context.User.FindFirst(ClaimTypes.NameIdentifier)?.Value;

        if (!long.TryParse(subject, out var userId))
        {
            return;
        }

        var isActive = await _db.Users
            .AnyAsync(user => user.Id == userId && user.IsActive);

        if (!isActive)
        {
            return;
        }

        var hasPermission = await _db.UserRoles
            .Where(userRole => userRole.UserId == userId)
            .Join(
                _db.RolePermissions,
                userRole => userRole.RoleId,
                rolePermission => rolePermission.RoleId,
                (userRole, rolePermission) => rolePermission)
            .AnyAsync(rolePermission =>
                rolePermission.IsAllowed &&
                requirement.PermissionCodes.Contains(
                    rolePermission.Permission.Code));

        if (hasPermission)
        {
            context.Succeed(requirement);
        }
    }
}

import { apiRequest } from "./apiClient";

const BASE_URL = "/api/bookings";

export const bookingService = {
    getAll(search = "", status = "") {
        const params = new URLSearchParams();

        if (search.trim()) {
            params.set("search", search.trim());
        }

        if (status && status !== "All") {
            params.set("status", status);
        }

        const query = params.toString();

        return apiRequest(`${BASE_URL}${query ? `?${query}` : ""}`);
    },

    getCustomers() {
        return apiRequest(`${BASE_URL}/customers`);
    },

    create(booking) {
        return apiRequest(BASE_URL, {
            method: "POST",
            body: JSON.stringify(booking)
        });
    }
};

const TOKEN_KEY = "qubriq_access_token";

export function getAccessToken() {
    return sessionStorage.getItem(TOKEN_KEY);
}

export function setAccessToken(token) {
    if (token) {
        sessionStorage.setItem(TOKEN_KEY, token);
    } else {
        sessionStorage.removeItem(TOKEN_KEY);
    }
}

export async function apiRequest(url, options = {}) {
    const headers = {
        ...options.headers
    };

    if (options.body && !headers["Content-Type"]) {
        headers["Content-Type"] = "application/json";
    }

    const token = getAccessToken();

    if (token) {
        headers.Authorization = `Bearer ${token}`;
    }

    const response = await fetch(url, {
        ...options,
        headers
    });

    const text = await response.text();
    let data = null;

    if (text) {
        try {
            data = JSON.parse(text);
        } catch {
            data = { message: text };
        }
    }

    if (!response.ok) {
        const validationErrors = data?.errors
            ? Object.values(data.errors).flat().join(" ")
            : "";

        const error = new Error(
            data?.message ||
            validationErrors ||
            data?.title ||
            `Request failed: HTTP ${response.status}`
        );

        error.status = response.status;
        error.data = data;
        throw error;
    }

    return data;
}
