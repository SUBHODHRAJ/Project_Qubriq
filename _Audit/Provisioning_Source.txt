using System.Data;
using System.Security.Cryptography;
using System.Text;
using Microsoft.AspNetCore.Authorization;

[ApiController]
[Route("api/auth/bootstrap")]
public class AdminBootstrapController : ControllerBase
{
    private readonly QubriqDbContext _db;
    private readonly IConfiguration _configuration;
    private readonly IWebHostEnvironment _environment;

    public AdminBootstrapController(
        QubriqDbContext db,
        IConfiguration configuration,
        IWebHostEnvironment environment)
    {
        _db = db;
        _configuration = configuration;
        _environment = environment;
    }

    [AllowAnonymous]
    [HttpPost("admin")]
    public async Task<IActionResult> CreateFirstAdmin(
        [FromBody] BootstrapAdminRequest request)
    {
        // Never expose bootstrap outside Development.
        if (!_environment.IsDevelopment())
        {
            return NotFound();
        }

        var configuredSecret = _configuration["Bootstrap:Secret"];
        var suppliedSecret = Request.Headers["X-Qubriq-Bootstrap-Key"].ToString();

        if (string.IsNullOrWhiteSpace(configuredSecret) ||
            string.IsNullOrWhiteSpace(suppliedSecret))
        {
            return Unauthorized(new
            {
                message = "Invalid bootstrap credentials."
            });
        }

        var expectedBytes = Encoding.UTF8.GetBytes(configuredSecret);
        var suppliedBytes = Encoding.UTF8.GetBytes(suppliedSecret);

        if (expectedBytes.Length != suppliedBytes.Length ||
            !CryptographicOperations.FixedTimeEquals(
                expectedBytes,
                suppliedBytes))
        {
            return Unauthorized(new
            {
                message = "Invalid bootstrap credentials."
            });
        }

        var fullName = request.FullName.Trim();
        var email = request.Email.Trim().ToLowerInvariant();

        if (string.IsNullOrWhiteSpace(fullName) ||
            string.IsNullOrWhiteSpace(email))
        {
            return BadRequest(new
            {
                message = "Full name and email are required."
            });
        }

        await using var transaction =
            await _db.Database.BeginTransactionAsync(
                IsolationLevel.Serializable);

        try
        {
            // Bootstrap is permitted only for a completely empty user table.
            var existingUser = await _db.Users.AnyAsync();

            if (existingUser)
            {
                await transaction.RollbackAsync();

                return Conflict(new
                {
                    message = "Bootstrap is disabled because a user account already exists."
                });
            }

            var adminRole = await _db.Roles
                .FirstOrDefaultAsync(x => x.Name == "Admin");

            if (adminRole == null)
            {
                await transaction.RollbackAsync();

                return StatusCode(500, new
                {
                    message = "Admin role is missing. Verify RBAC seed data."
                });
            }

            var user = new User
            {
                FullName = fullName,
                Email = email,
                EmailVerified = true,
                PasswordHash = null,
                IsActive = true,
                CreatedAtUtc = DateTime.UtcNow
            };

            var passwordHasher = new PasswordHasher<User>();

            user.PasswordHash = passwordHasher.HashPassword(
                user,
                request.Password);

            _db.Users.Add(user);

            await _db.SaveChangesAsync();

            var userRole = new UserRole
            {
                UserId = user.Id,
                RoleId = adminRole.Id,
                AssignedAtUtc = DateTime.UtcNow
            };

            _db.UserRoles.Add(userRole);

            await _db.SaveChangesAsync();
            await transaction.CommitAsync();

            return Created(
                "/api/auth/login",
                new
                {
                    message = "Initial Admin account created successfully.",
                    userId = user.Id,
                    fullName = user.FullName,
                    email = user.Email,
                    role = "Admin",
                    nextStep = "Sign in using POST /api/auth/login."
                });
        }
        catch (DbUpdateException)
        {
            await transaction.RollbackAsync();

            return Conflict(new
            {
                message = "Account creation conflicted with existing database data."
            });
        }
    }
}

using System.Data;
using System.Security.Claims;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Identity;
using Microsoft.AspNetCore.Mvc;

using Qubriq.API.Data;

namespace Qubriq.API.Controllers;

[ApiController]
[Route("api/admin/customers")]
[Authorize(Policy = "AdminCustomerProvision")]
public class AdminCustomersController : ControllerBase
{
    private readonly QubriqDbContext _db;
    private readonly PasswordHasher<User> _passwordHasher;

    public AdminCustomersController(QubriqDbContext db)
    {
        _db = db;
        _passwordHasher = new PasswordHasher<User>();
    }

    private long? GetAuthenticatedUserId()
    {
        var subject =
            User.FindFirst("sub")?.Value ??
            User.FindFirst(ClaimTypes.NameIdentifier)?.Value;

        if (long.TryParse(subject, out var userId))
        {
            return userId;
        }

        return null;
    }

    // POST: api/admin/customers/{customerId}/account
    [HttpPost("{customerId:long}/account")]
    public async Task<IActionResult> CreateCustomerAccount(
        long customerId,
        [FromBody] CreateCustomerAccountRequest request)
    {
        var adminUserId = GetAuthenticatedUserId();

        if (!adminUserId.HasValue)
        {
            return Unauthorized();
        }

        var email = request.Email.Trim().ToLowerInvariant();
        var fullName = request.FullName.Trim();

        if (string.IsNullOrWhiteSpace(fullName) ||
            string.IsNullOrWhiteSpace(email) ||
            string.IsNullOrWhiteSpace(request.Password))
        {
            return BadRequest(new
            {
                message = "Full name, email and password are required."
            });
        }

        if (request.Password.Length < 12 ||
            request.Password.Length > 128)
        {
            return BadRequest(new
            {
                message = "Password must contain between 12 and 128 characters."
            });
        }

        await using var transaction =
            await _db.Database.BeginTransactionAsync(
                IsolationLevel.Serializable);

        var customer = await _db.Customers
            .FirstOrDefaultAsync(x => x.Id == customerId);

        if (customer == null)
        {
            return NotFound(new
            {
                message = "Customer record was not found."
            });
        }

        if (!customer.IsActive)
        {
            return BadRequest(new
            {
                message = "An inactive customer cannot receive a login account."
            });
        }

        if (customer.UserId.HasValue)
        {
            return Conflict(new
            {
                message = "This customer already has a linked user account.",
                customerId = customer.Id
            });
        }

        var existingUser = await _db.Users
            .AnyAsync(x => x.Email == email);

        if (existingUser)
        {
            return Conflict(new
            {
                message = "An account with this email address already exists."
            });
        }

        var customerRole = await _db.Roles
            .FirstOrDefaultAsync(x => x.Name == "BookingCustomer");

        if (customerRole == null)
        {
            return StatusCode(500, new
            {
                message = "BookingCustomer role is missing. Check RBAC seed data."
            });
        }

        var user = new User
        {
            FullName = fullName,
            Email = email,
            IsActive = true,
            EmailVerified = false,
            RecipientType = null,
            CreatedAtUtc = DateTime.UtcNow,
            UpdatedAtUtc = DateTime.UtcNow
        };

        user.PasswordHash = _passwordHasher.HashPassword(
            user,
            request.Password);

        _db.Users.Add(user);

        await _db.SaveChangesAsync();

        var userRole = new UserRole
        {
            UserId = user.Id,
            RoleId = customerRole.Id,
            AssignedAtUtc = DateTime.UtcNow
        };

        _db.UserRoles.Add(userRole);

        customer.UserId = user.Id;

        var auditLog = new AuditLog
        {
            UserId = adminUserId.Value,
            Action = "CUSTOMER_ACCOUNT_PROVISIONED",
            EntityType = "Customer",
            EntityId = customer.Id.ToString(),
            Details =
                $"Created BookingCustomer account. UserId={user.Id}; Email={email}",
            CreatedAtUtc = DateTime.UtcNow
        };

        _db.AuditLogs.Add(auditLog);

        await _db.SaveChangesAsync();

        await transaction.CommitAsync();

        return Ok(new
        {
            message = "Customer account created and linked successfully.",
            customer = new
            {
                customer.Id,
                customer.CompanyName,
                customer.UserId
            },
            account = new
            {
                user.Id,
                user.FullName,
                user.Email,
                role = customerRole.Name,
                user.IsActive,
                user.EmailVerified
            }
        });
    }
}

using System.Data;
using System.Security.Claims;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Identity;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using Qubriq.API.Data;
using Qubriq.API.Models;

namespace Qubriq.API.Controllers;

[ApiController]
[Route("api/admin/recipients")]
[Authorize(Roles = "Admin")]
public class AdminRecipientsController : ControllerBase
{
    private readonly QubriqDbContext _db;
    private readonly PasswordHasher<User> _hasher = new();

    public AdminRecipientsController(QubriqDbContext db)
    {
        _db = db;
    }

    private long? AdminId()
    {
        var value = User.FindFirst("sub")?.Value ??
                    User.FindFirst(ClaimTypes.NameIdentifier)?.Value;

        return long.TryParse(value, out var id) ? id : null;
    }

    private async Task<bool> HasPermission(string code)
    {
        var id = AdminId();
        if (!id.HasValue) return false;

        return await _db.UserRoles
            .Where(x => x.UserId == id.Value)
            .Join(_db.RolePermissions,
                x => x.RoleId,
                x => x.RoleId,
                (_, permission) => permission)
            .AnyAsync(x =>
                x.IsAllowed && x.Permission.Code == code);
    }

    public sealed class CreateRecipientRequest
    {
        public string FullName { get; set; } = "";
        public string Email { get; set; } = "";
        public string Password { get; set; } = "";
        public string RecipientType { get; set; } = "";
        public long? DriverId { get; set; }
    }

    [HttpPost]
    public async Task<IActionResult> CreateRecipient(
        [FromBody] CreateRecipientRequest request)
    {
        if (!await HasPermission("users.create") ||
            !await HasPermission("users.assign_role"))
            return Forbid();

        if (request == null)
            return BadRequest(new { message = "Recipient details are required." });

        var name = request.FullName?.Trim() ?? "";
        var email = request.Email?.Trim().ToLowerInvariant() ?? "";
        var password = request.Password;
        var type = request.RecipientType?.Trim() ?? "";

        if (string.IsNullOrWhiteSpace(name) || name.Length > 150)
            return BadRequest(new { message = "Name is required and must not exceed 150 characters." });

        if (string.IsNullOrWhiteSpace(email) || email.Length > 254)
            return BadRequest(new { message = "A valid email address is required." });

        try
        {
            var parsedEmail = new System.Net.Mail.MailAddress(email);

            if (!string.Equals(
                parsedEmail.Address,
                email,
                StringComparison.OrdinalIgnoreCase))
            {
                return BadRequest(new { message = "A valid email address is required." });
            }
        }
        catch (FormatException)
        {
            return BadRequest(new { message = "A valid email address is required." });
        }

        if (string.IsNullOrEmpty(password) ||
            password.Length < 12 ||
            password.Length > 128)
        {
            return BadRequest(new { message = "Password must be 12-128 characters." });
        }

        if (type != "DeliveryPersonnel" && type != "FinalRecipient")
            return BadRequest(new { message = "Invalid recipient subtype." });

        if (type == "DeliveryPersonnel" && !request.DriverId.HasValue)
            return BadRequest(new { message = "DriverId is required for DeliveryPersonnel." });

        if (type == "FinalRecipient" && request.DriverId.HasValue)
            return BadRequest(new { message = "FinalRecipient cannot be linked to a Driver." });

        await using var tx = await _db.Database.BeginTransactionAsync(
            IsolationLevel.Serializable);

        if (await _db.Users.AnyAsync(x => x.Email == email))
            return Conflict(new { message = "Email already exists." });

        Driver? driver = null;

        if (type == "DeliveryPersonnel")
        {
            driver = await _db.Drivers
                .FirstOrDefaultAsync(x => x.Id == request.DriverId.GetValueOrDefault());

            if (driver == null)
                return NotFound(new { message = "Driver not found." });

            if (driver.UserId.HasValue)
                return Conflict(new { message = "Driver already has a linked user." });
        }

        var role = await _db.Roles
            .FirstOrDefaultAsync(x => x.Name == "ShipmentRecipient");

        if (role == null)
            return StatusCode(500, new { message = "ShipmentRecipient role is missing." });

        var user = new User
        {
            FullName = name,
            Email = email,
            RecipientType = type,
            IsActive = true,
            EmailVerified = false,
            CreatedAtUtc = DateTime.UtcNow,
            UpdatedAtUtc = DateTime.UtcNow
        };

        user.PasswordHash = _hasher.HashPassword(user, request.Password);

        _db.Users.Add(user);
        await _db.SaveChangesAsync();

        _db.UserRoles.Add(new UserRole
        {
            UserId = user.Id,
            RoleId = role.Id,
            AssignedAtUtc = DateTime.UtcNow
        });

        if (driver != null)
            driver.UserId = user.Id;

        _db.AuditLogs.Add(new AuditLog
        {
            UserId = AdminId(),
            Action = "RECIPIENT_ACCOUNT_PROVISIONED",
            EntityType = "User",
            EntityId = user.Id.ToString(),
            Details = $"RecipientType={type}; DriverId={driver?.Id}",
            CreatedAtUtc = DateTime.UtcNow
        });

        await _db.SaveChangesAsync();
        await tx.CommitAsync();

        return Created("", new
        {
            user.Id,
            user.FullName,
            user.Email,
            user.RecipientType,
            user.IsActive,
            DriverId = driver?.Id
        });
    }
}

[ApiController]
[Route("api/fleet")]
[Authorize(Roles = "Admin,OperationsDispatcher")]
public class FleetController : ControllerBase
{
    private readonly QubriqDbContext _db;

    public FleetController(QubriqDbContext db)
    {
        _db = db;
    }

    private long? UserId()
    {
        var value = User.FindFirst("sub")?.Value
            ?? User.FindFirst(ClaimTypes.NameIdentifier)?.Value;
        return long.TryParse(value, out var id) ? id : null;
    }

    private async Task<bool> Allowed(string permission)
    {
        var id = UserId();
        if (!id.HasValue) return false;

        if (!await _db.Users.AnyAsync(x => x.Id == id.Value && x.IsActive))
            return false;

        return await _db.UserRoles
            .Where(x => x.UserId == id.Value)
            .Join(_db.RolePermissions,
                ur => ur.RoleId,
                rp => rp.RoleId,
                (_, rp) => rp)
            .AnyAsync(x => x.IsAllowed && x.Permission.Code == permission);
    }

    // VEHICLES

    [HttpGet("vehicles")]
    public async Task<IActionResult> GetVehicles()
    {
        if (!await Allowed("fleet.vehicles.manage")) return Forbid();

        var items = await _db.Vehicles
            .AsNoTracking()
            .OrderBy(x => x.RegistrationNumber)
            .Select(x => new
            {
                x.Id,
                x.RegistrationNumber,
                x.VehicleType,
                x.CapacityKg,
                x.Status
            }).ToListAsync();

        return Ok(items);
    }

    [HttpPost("vehicles")]
    public async Task<IActionResult> CreateVehicle(VehicleRequest request)
    {
        if (!await Allowed("fleet.vehicles.manage")) return Forbid();

        if (string.IsNullOrWhiteSpace(request.RegistrationNumber) ||
            string.IsNullOrWhiteSpace(request.VehicleType) ||
            request.RegistrationNumber.Trim().Length > 30 ||
            request.VehicleType.Trim().Length > 80 ||
            request.CapacityKg <= 0)
            return BadRequest(new { message = "Enter a valid registration, vehicle type and positive capacity." });

        var registration = request.RegistrationNumber.Trim().ToUpperInvariant();

        if (await _db.Vehicles.AnyAsync(x => x.RegistrationNumber == registration))
            return Conflict(new { message = "Registration number already exists." });

        var vehicle = new Vehicle
        {
            RegistrationNumber = registration,
            VehicleType = request.VehicleType.Trim(),
            CapacityKg = request.CapacityKg,
            Status = "Available"
        };

        _db.Vehicles.Add(vehicle);
        await _db.SaveChangesAsync();

        return CreatedAtAction(nameof(GetVehicles), new { },
            new { vehicle.Id, vehicle.RegistrationNumber, vehicle.Status });
    }

    [HttpPatch("vehicles/{id:long}/status")]
    public async Task<IActionResult> UpdateVehicleStatus(long id, StatusRequest request)
    {
        if (!await Allowed("fleet.vehicles.manage")) return Forbid();

        if (request.Status is not ("Available" or "Maintenance" or "Inactive"))
            return BadRequest(new { message = "Status must be Available, Maintenance or Inactive." });

        var vehicle = await _db.Vehicles.FirstOrDefaultAsync(x => x.Id == id);
        if (vehicle == null) return NotFound();

        var active = await _db.ShipmentAssignments.AnyAsync(
            x => x.VehicleId == id && x.Status == "Active");

        if (active && request.Status != "Available")
            return Conflict(new { message = "Release the active assignment before changing vehicle status." });

        vehicle.Status = request.Status;
        await _db.SaveChangesAsync();
        return Ok(new { vehicle.Id, vehicle.Status });
    }

        var items = await _db.ShipmentAssignments

            .Include(x => x.Shipment)

            .Include(x => x.Vehicle)

    [HttpPost("assignments")]
    public async Task<IActionResult> Assign(AssignmentRequest request)
    {
        if (!await Allowed("fleet.assignments.manage")) return Forbid();

        if (request.ShipmentId <= 0 || request.DriverId <= 0 || request.VehicleId <= 0)
            return BadRequest(new { message = "Select a shipment, driver and vehicle." });

        await using var tx = await _db.Database.BeginTransactionAsync(IsolationLevel.Serializable);

        var shipment = await _db.Shipments
            .FirstOrDefaultAsync(x => x.Id == request.ShipmentId);

        if (shipment == null) return NotFound(new { message = "Shipment not found." });

        if (shipment.Status != "Confirmed")
            return Conflict(new { message = "Only Confirmed bookings can be assigned." });

        var driver = await _db.Drivers.FirstOrDefaultAsync(x => x.Id == request.DriverId);
        var vehicle = await _db.Vehicles.FirstOrDefaultAsync(x => x.Id == request.VehicleId);

        if (driver == null || vehicle == null)
            return BadRequest(new { message = "Driver or vehicle not found." });

        if (driver.Status != "Available" || vehicle.Status != "Available")
            return Conflict(new { message = "Driver and vehicle must both be Available." });

        if (shipment.TotalWeightKg.HasValue &&
            shipment.TotalWeightKg.Value > vehicle.CapacityKg)
            return Conflict(new { message = "Shipment weight exceeds vehicle capacity." });

        var shipmentAssigned = await _db.ShipmentAssignments.AnyAsync(
            x => x.ShipmentId == request.ShipmentId && x.Status == "Active");

        var driverAssigned = await _db.ShipmentAssignments.AnyAsync(
            x => x.DriverId == request.DriverId && x.Status == "Active");

        var vehicleAssigned = await _db.ShipmentAssignments.AnyAsync(
            x => x.VehicleId == request.VehicleId && x.Status == "Active");

        if (shipmentAssigned || driverAssigned || vehicleAssigned)
            return Conflict(new { message = "Shipment, driver or vehicle already has an active assignment." });

        var assignment = new ShipmentAssignment
        {
            ShipmentId = shipment.Id,
            DriverId = driver.Id,
            VehicleId = vehicle.Id,
            AssignedAtUtc = DateTime.UtcNow,
            Status = "Active"
        };

        _db.ShipmentAssignments.Add(assignment);
        shipment.Status = "Assigned";

        _db.AuditLogs.Add(new AuditLog
        {
            UserId = UserId(),
            Action = "SHIPMENT_DRIVER_ASSIGNED",
            EntityType = "Shipment",
            EntityId = shipment.Id.ToString(),
            Details = $"DriverId={driver.Id}; VehicleId={vehicle.Id}",
            CreatedAtUtc = DateTime.UtcNow
        });

        await _db.SaveChangesAsync();
        await tx.CommitAsync();

        return Ok(new
        {
            assignment.Id,
            assignment.ShipmentId,
            assignment.DriverId,
            assignment.VehicleId,
            shipment.Status
        });
    }
}

using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Qubriq.API.DTOs.Auth;
using Qubriq.API.Services;

namespace Qubriq.API.Controllers;

[ApiController]
[Route("api/auth")]
public class AuthController : ControllerBase
{
    private readonly AuthService _authService;

    public AuthController(AuthService authService)
    {
        _authService = authService;
    }

    [AllowAnonymous]
    [HttpPost("login")]
    public async Task<IActionResult> Login(
        [FromBody] LoginRequest request)
    {
        var result = await _authService.LoginAsync(request);

        if (result == null)
        {
            return Unauthorized(new
            {
                message = "Invalid email or password, or the account is inactive."

        var canReadAll = await HasPermissionAsync(

        Customer? customer = null;

        if (canReadAll)
        {
            if (!request.CustomerId.HasValue ||
                request.CustomerId.Value <= 0)
            {
                return BadRequest(new
                {

                .FirstOrDefaultAsync(x =>
                    x.Id == request.CustomerId.Value &&
                    x.IsActive);
        }
        else
        {
            // Never trust a customer ID submitted by an external customer.
            // Resolve the customer's own profile from the authenticated user.
            customer = await _db.Customers
                .FirstOrDefaultAsync(x =>
                    x.UserId == userId.Value &&
                    x.IsActive);
        }

        if (customer == null)

            return BadRequest(new
            {
                message = canReadAll
                    ? "Select an active customer before creating a booking."
                    : "Your account does not have an active customer profile. Contact Qubriq support."
            });

                message = "A valid pickup date is required."

        {
            return BadRequest(new

                message = "Expected delivery cannot be earlier than pickup."
            });

    [HttpPatch("{id:long}/status")]
    public async Task<IActionResult> UpdateStatus(
        long id,
        [FromBody] StatusRequest request)
    {
        var user = await ActiveRecipient();
        if (user == null || user.RecipientType != "DeliveryPersonnel")
            return Forbid();

        if (!await HasPermission(user.Id, "delivery.status.update"))
            return Forbid();

        var shipment = await Scoped(user)
            .FirstOrDefaultAsync(x => x.Id == id);

        if (shipment == null) return NotFound();

        var allowed = new Dictionary<string, string[]>
        {
            ["Assigned"] = new[] { "PickedUp" },
            ["PickedUp"] = new[] { "InTransit" },
            ["InTransit"] = new[] { "Delivered" }
        };

        if (string.IsNullOrWhiteSpace(request.Status) ||
            !allowed.TryGetValue(shipment.Status, out var next))
            return BadRequest(new { message = "Invalid shipment status transition." });

        var canonicalStatus = next.FirstOrDefault(
            x => string.Equals(x, request.Status.Trim(), StringComparison.OrdinalIgnoreCase));

        if (canonicalStatus == null)
            return BadRequest(new { message = "Invalid shipment status transition." });

        var oldStatus = shipment.Status;
        shipment.Status = canonicalStatus;

        _db.AuditLogs.Add(new AuditLog
        {
            UserId = user.Id,
            Action = "SHIPMENT_STATUS_CHANGED",
            EntityType = "Shipment",
            EntityId = shipment.Id.ToString(),
            Details = $"From={oldStatus}; To={request.Status}",
            CreatedAtUtc = DateTime.UtcNow
        });

        await _db.SaveChangesAsync();

        return Ok(new { shipment.Id, shipment.TrackingNumber, shipment.Status });
    }
