  358:     private static bool Intersects(Box a, Box b)
  359:     {
  360:         return a.X < b.Right - Epsilon &&
  361:                a.Right > b.X + Epsilon &&
  362:                a.Y < b.Back - Epsilon &&
  363:                a.Back > b.Y + Epsilon &&
  364:                a.Z < b.Top - Epsilon &&
  365:                a.Top > b.Z + Epsilon;

  368:     private static bool HasStableSupport(
  369:         Box candidate,
  370:         List<Box> placements)
  371:     {
  372:         if (candidate.Z <= Epsilon)
  373:             return true;
  374: 
  375:         var footprintArea = candidate.Length * candidate.Width;
  376:         if (footprintArea <= 0)
  377:             return false;
  378: 
  379:         decimal supportedArea = 0;
  380: 
  381:         foreach (var support in placements)
  382:         {
  383:             if (Math.Abs(support.Top - candidate.Z) > Epsilon)
  384:                 continue;
  385: 
  386:             var overlapLength = Math.Max(
  387:                 0,
  388:                 Math.Min(candidate.Right, support.Right) -
  389:                 Math.Max(candidate.X, support.X));
  390: 
  391:             var overlapWidth = Math.Max(
  392:                 0,
  393:                 Math.Min(candidate.Back, support.Back) -
  394:                 Math.Max(candidate.Y, support.Y));
  395: 
  396:             supportedArea += overlapLength * overlapWidth;
  397:         }
  398: 
  399:         return supportedArea / footprintArea >= SupportThreshold;
  400:     }

  402:     private static bool WouldLoadOnFragile(

  410:             candidate.Right > existing.X + Epsilon &&
  411:             candidate.Y < existing.Back - Epsilon &&
